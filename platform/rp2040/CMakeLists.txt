cmake_minimum_required(VERSION 3.20)

set(PICO_BOARD waveshare_rp2040_zero CACHE STRING "Gridpico RP2040 bench board")
if(NOT PICO_SDK_PATH)
    set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif()
if(NOT PICO_TINYUSB_PATH)
    set(PICO_TINYUSB_PATH "$ENV{PICO_TINYUSB_PATH}")
endif()
if(NOT EXISTS "${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
    message(FATAL_ERROR "Set PICO_SDK_PATH to the installed Pico SDK.")
endif()
if(NOT EXISTS "${PICO_TINYUSB_PATH}/src/tusb_option.h")
    message(FATAL_ERROR "Set PICO_TINYUSB_PATH to the installed TinyUSB 0.21.0 tree.")
endif()

include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(gridpico_rp2040 LANGUAGES C CXX ASM)
if(NOT PICO_PLATFORM STREQUAL "rp2040")
    message(FATAL_ERROR "The Gridpico recovery probes require RP2040.")
endif()
pico_sdk_init()

function(add_recovery_probe target)
    add_executable("${target}" recovery.c bootsel_line.c)
    target_include_directories("${target}" PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries("${target}" PRIVATE pico_stdlib pico_bootrom hardware_watchdog tinyusb_device)
    set_target_properties("${target}" PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
    pico_enable_stdio_usb("${target}" 0)
    pico_enable_stdio_uart("${target}" 0)
    pico_set_binary_type("${target}" no_flash)
    pico_add_extra_outputs("${target}")
endfunction()

add_recovery_probe(gridpico_rp2040_recovery)
option(GRIDPICO_RP2040_WATCHDOG_PROBE "Build the separate RAM watchdog fault probe" OFF)
if(GRIDPICO_RP2040_WATCHDOG_PROBE)
    add_recovery_probe(gridpico_rp2040_watchdog_probe)
    target_compile_definitions(gridpico_rp2040_watchdog_probe PRIVATE GRIDPICO_RP2040_WATCHDOG_PROBE)
endif()
